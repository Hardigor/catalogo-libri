        const showStatus = (message, type = 'success', duration = 5000) => {

            clearTimeout(statusTimeout);

            const icons = {

                success: 'fa-check-circle',

                error: 'fa-exclamation-triangle',

                loading: 'fa-sync fa-spin'

            };

            dom.statusMessage.innerHTML = `<i class="fas ${icons[type]}"></i> ${message}`;

            dom.statusMessage.className = `status-message status-${type}`;

            dom.statusMessage.classList.add('show');

            if (duration !== null) {

                statusTimeout = setTimeout(() => dom.statusMessage.classList.remove('show'), duration);

            }

        };


        const hideStatus = () => dom.statusMessage.classList.remove('show');


        const debounce = (func, delay) => {

            let timeout;

            return (...args) => {

                clearTimeout(timeout);

                timeout = setTimeout(() => func.apply(this, args), delay);

            };

        };

        const getBookDetails = async (author, title) => {

            // Usa Open Library API (Gratuita, nessuna Key necessaria)

            const searchQuery = `title:${encodeURIComponent(title)} author:${encodeURIComponent(author)}`;

            const url = `https://openlibrary.org/search.json?q=${searchQuery}&limit=1`;

            

            try {

                const response = await fetch(url);

                if (!response.ok) throw new Error("Errore Network Open Library");

                const data = await response.json();


                if (data.numFound === 0 || !data.docs || data.docs.length === 0) {

                    return null;

                }


                const book = data.docs[0];

                const coverId = book.cover_i;

                const coverUrl = coverId 

                    ? `https://covers.openlibrary.org/b/id/${coverId}-L.jpg` 

                    : null;


                // Open Library restituisce 'subject' che sono i tag/generi/argomenti

                const tags = book.subject ? book.subject.slice(0, 10) : []; // Prendiamo i primi 10


                return {

                    cover: coverUrl,

                    tags: tags,

                    key: book.key // Identificativo Open Library

                };


            } catch (error) {

                console.error("Errore Open Library:", error);

                return null;

            }

        };

        

        const renderBooks = () => {

            filteredBooks.sort((a, b) => {

                const valA = a[sortState.field];

                const valB = b[sortState.field];

                let comparison = typeof valA === 'number' && typeof valB === 'number' 

                    ? valA - valB 

                    : String(valA).localeCompare(String(valB), 'it', { sensitivity: 'base' });

                return comparison * sortState.direction;

            });


            const totalPages = Math.ceil(filteredBooks.length / itemsPerPage);

            const startIndex = (currentPage - 1) * itemsPerPage;

            const paginatedBooks = filteredBooks.slice(startIndex, startIndex + itemsPerPage);


            dom.bookList.innerHTML = '';

            if (paginatedBooks.length === 0) {

                dom.bookList.innerHTML = `<tr><td colspan="4" class="empty-state-cell"><i class="fas fa-book"></i>Nessun libro trovato.</td></tr>`;

            } else {

                paginatedBooks.forEach(book => {

                    const row = document.createElement('tr');

                    row.className = 'fade-in';

                    row.setAttribute('data-author', book.author);

                    row.setAttribute('data-title', book.title);

                    row.innerHTML = `<td>${book.title}</td><td>${book.author}</td><td>${book.year || ''}</td><td>${book.genre}</td>`;

                    dom.bookList.appendChild(row);

                });

            }


            updateSortIndicators();

            updatePagination(totalPages);

            updateStats();

        };


        const applyFiltersAndSearch = () => {

            const searchTerm = dom.searchInput.value.toLowerCase();

            const genreFilter = dom.filterGenre.value;

            const yearFilter = dom.filterYear.value;

            filteredBooks = books.filter(book => {

                const matchesSearch = searchTerm ? 

                    ['title', 'author', 'genre'].some(field => String(book[field]).toLowerCase().includes(searchTerm)) || 

                    String(book.year).includes(searchTerm) : true;

                const matchesGenre = genreFilter ? book.genre === genreFilter : true;

                let matchesYear = true;

                if (yearFilter) {

                    const year = book.year;

                    const ranges = {

                        '2020s': [2020, new Date().getFullYear()], '2010s': [2010, 2019], '2000s': [2000, 2009],

                        '1990s': [1990, 1999], '1980s': [1980, 1989], '1970s': [1970, 1979], '1960s': [1960, 1969],

                        '1950s': [1950, 1959], 'pre1950': [0, 1949]

                    };

                    if (ranges[yearFilter]) {

                        matchesYear = year >= ranges[yearFilter][0] && year <= ranges[yearFilter][1];

                    }

                }

                return matchesSearch && matchesGenre && matchesYear;

            });

            currentPage = 1;

            renderBooks();

        };

        const parseCsv = (text) => {

            const lines = text.trim().split('\n').slice(1);

            return lines.map(line => {

                // CSV Atteso: Title, Author, Year, Genre

                const [title, author, year, genre] = line.split(',').map(v => v.trim().replace(/^"|"$/g, ''));

                return { title, author, year: parseInt(year) || 0, genre };

            }).filter(b => b.title && b.author);

        };


        const importFromUrl = async (isAuto = false) => {

            const url = dom.sheetsUrlInput.value.trim();

            if (!url) { if (!isAuto) showStatus('Inserisci un URL valido.', 'error'); return; }

            const loadingMessage = isAuto ? 'Aggiornamento automatico...' : 'Importazione...';

            showStatus(loadingMessage, 'loading', null);

            dom.loader.style.display = 'block';

            try {

                const response = await fetch(url + `&t=${Date.now()}`);

                if (!response.ok) throw new Error(`Errore HTTP: ${response.status}`);

                const csvText = await response.text();

                books = parseCsv(csvText);

                localStorage.setItem('books', JSON.stringify(books));

                applyFiltersAndSearch();

                showStatus(`Importati ${books.length} Libri con successo!`, 'success');

            } catch (error) {

                showStatus(`Errore importazione: ${error.message}`, 'error');

                loadFromStorage();

            } finally {

                dom.loader.style.display = 'none';

            }

        };

        const init = () => {
